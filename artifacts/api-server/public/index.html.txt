<body>
  <h1>🔍 Decompiler</h1>
  <div id="drop">
    <p><strong>Trage un .jar aici</strong> sau click pentru a alege</p>
    <p style="font-size:.85rem">.jar / .class / .zip · max 200MB</p>
    <input id="file-input" type="file" accept=".jar,.class,.zip" />
  </div>

  <div id="status">
    <div id="stage">Se pornește…</div>
    <div id="bar-track"><div id="bar"></div></div>
  </div>

  <div id="result"></div>

  <script>
    const drop = document.getElementById('drop');
    const fileInput = document.getElementById('file-input');
    const statusBox = document.getElementById('status');
    const stageEl = document.getElementById('stage');
    const bar = document.getElementById('bar');
    const resultBox = document.getElementById('result');

    // Rough stage → progress-bar mapping just for visual feedback (worker doesn't emit %).
    const STAGE_PROGRESS = [
      'Checking dependencies', 'Extracting', 'Detecting obfuscation',
      'Deobfuscating', 'Repacking', 'Decompiling', 'Merging', 'Packaging',
    ];

    function progressForStage(stage) {
      const idx = STAGE_PROGRESS.findIndex(s => stage.toLowerCase().includes(s.toLowerCase()));
      if (idx === -1) return 10;
      return Math.round(((idx + 1) / STAGE_PROGRESS.length) * 100);
    }

    ['dragenter', 'dragover'].forEach(evt =>
      drop.addEventListener(evt, e => { e.preventDefault(); drop.classList.add('drag'); }));
    ['dragleave', 'drop'].forEach(evt =>
      drop.addEventListener(evt, e => { e.preventDefault(); drop.classList.remove('drag'); }));
    drop.addEventListener('drop', e => {
      const file = e.dataTransfer.files[0];
      if (file) upload(file);
    });
    drop.addEventListener('click', () => fileInput.click());
    fileInput.addEventListener('change', () => {
      if (fileInput.files[0]) upload(fileInput.files[0]);
    });

    async function upload(file) {
      resultBox.style.display = 'none';
      resultBox.innerHTML = '';
      statusBox.style.display = 'block';
      stageEl.textContent = 'Se încarcă…';
      bar.style.width = '3%';

      const form = new FormData();
      form.append('file', file);

      let res;
      try {
        res = await fetch('/api/jobs', { method: 'POST', body: form });
      } catch (e) {
        return showError('Nu am putut contacta serverul: ' + e.message);
      }
      if (!res.ok) {
        const body = await res.json().catch(() => ({}));
        return showError(body.error || `Upload eșuat (${res.status})`);
      }
      const { id } = await res.json();
      poll(id);
    }

    async function poll(id) {
      const res = await fetch(`/api/jobs/${id}`);
      if (!res.ok) return showError('Job negăsit.');
      const job = await res.json();

      if (job.status === 'queued' && job.queuePosition > 0) {
        stageEl.textContent = `În așteptare — locul ${job.queuePosition} în coadă…`;
        bar.style.width = '3%';
      } else {
        stageEl.textContent = job.stage;
        bar.style.width = progressForStage(job.stage || '') + '%';
      }

      if (job.status === 'done') {
        bar.style.width = '100%';
        showResult(id, job);
      } else if (job.status === 'failed') {
        showError(job.error || 'Eroare necunoscută în pipeline.');
      } else {
        setTimeout(() => poll(id), 1500);
      }
    }

    function showResult(id, job) {
      resultBox.style.display = 'block';
      resultBox.innerHTML = `
        <a class="btn" href="/api/jobs/${id}/download">⬇ Descarcă rezultatul (${job.fileCount ?? '?'} fișiere)</a>
        <div class="summary">${(job.summary || '').replace(/</g, '&lt;')}</div>
      `;
    }

    function showError(msg) {
      statusBox.style.display = 'none';
      resultBox.style.display = 'block';
      resultBox.innerHTML = `<div class="err">❌ ${msg.replace(/</g, '&lt;')}</div>`;
    }
  </script>
